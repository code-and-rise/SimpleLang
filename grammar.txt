<program> := <statement_list>

<statement_list> := statement <statement_list>
<statement_list> := $

<statement> := <assignment>
<statement> := <if_stmt>
<statement> := <for_stmt>
<statement> := <print_stmt>
<statement> := <while_stmt>

<assignment> := KW_VAR IDN OP_ASSIGN <expr>

<if_stmt> := KW_IF <expr> <block> <if_stmt_tail>

<if_stmt_tail> := KW_ELSE <block>
<if_stmt_tail> := KW_ELSE <if_stmt>
<if_stmt_tail> := $

<for_stmt> := KW_FOR IDN KW_FROM <expr> KW_TO <expr> <block>

<print_stmt> := KW_PRINT <expr>

<while_stmt> := KW_WHILE <expr> <block>

<block> := L_BRACET <statement_list> R_BRACET

<expr> := <comparison>
<comparison> := <operand> (<rel_op> <operand>)*
<operand> := <term> ((OP_PLUS | OP_MINUS) <term>)*
<term> := <factor> ((OP_STAR | OP_SLASH) <factor>)*
<factor> := IDN | CONST_NUM | CONST_STRING | L_PARENT <expr> R_PARENT
<rel_op> := OP_EQUAL | OP_NOT_EQUAL | OP_LESS | OP_GREATER | OP_LESS_EQUAL | OP_GREATER_EQUAL
